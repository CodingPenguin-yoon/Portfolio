```ts
test('slide bodies stay above their footers without overflow', async ({ page }) => {
  await page.emulateMedia({ media: 'print' });
  await page.goto('/portfolio');
  const results = await page.locator('[data-portfolio-page]').evaluateAll((slides) =>
    slides.map((slide) => {
      const body = slide.querySelector<HTMLElement>('.slide-body');
      const footer = slide.querySelector<HTMLElement>('.page-footer');
      if (!body || !footer) throw new Error('Slide geometry is incomplete');
      const bodyBox = body.getBoundingClientRect();
      const footerBox = footer.getBoundingClientRect();
      return {
        noVerticalCollision: bodyBox.bottom <= footerBox.top + 0.5,
        noOverflow: body.scrollHeight <= body.clientHeight + 1 && body.scrollWidth <= body.clientWidth + 1,
      };
    })
  );
  expect(results.every(({ noVerticalCollision, noOverflow }) => noVerticalCollision && noOverflow)).toBe(true);
});
```

```ts
test('non-endpoint nodes do not intersect connector segments', async ({ page }) => {
  await page.emulateMedia({ media: 'print' });
  await page.goto('/portfolio');
  const violations = await page.locator('[data-diagram]').evaluateAll((diagrams) =>
    diagrams.flatMap((diagram) => {
      const nodes = Array.from(diagram.querySelectorAll<HTMLElement>('[data-diagram-node]'));
      const segments = Array.from(diagram.querySelectorAll<HTMLElement>('[data-connector-segment]'));
      return segments.flatMap((segment) => {
        const segmentBox = segment.getBoundingClientRect();
        const from = segment.dataset.fromNode;
        const to = segment.dataset.toNode;
        return nodes
          .filter((node) => ![from, to].includes(node.dataset.diagramNode))
          .filter((node) => {
            const nodeBox = node.getBoundingClientRect();
            return !(
              segmentBox.right <= nodeBox.left + 1 ||
              segmentBox.left >= nodeBox.right - 1 ||
              segmentBox.bottom <= nodeBox.top + 1 ||
              segmentBox.top >= nodeBox.bottom - 1
            );
          })
          .map((node) => `${segment.dataset.connectorSegment}:${node.dataset.diagramNode}`);
      });
    })
  );
  expect(violations).toEqual([]);
});
```
